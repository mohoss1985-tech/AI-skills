### Tests

impayes_test.py
```python
import os, uuid, pytest, httpx

BASE = os.environ["API_BASE_URL"]

@pytest.fixture
def analyste():
    return httpx.Client(base_url=BASE, headers={"Authorization": f"Bearer {os.environ['ANALYSTE_TOKEN']}"})

def emettre(api, op, motif="DOUBLON", montant=None, cle=None, **extra):
    corps = {"reference_operation": op.arn, "motif": motif, "montant": montant or op.montant, "devise": op.devise, **extra}
    return api.post("/v1/impayes", json=corps, headers={"Idempotency-Key": cle or str(uuid.uuid4())})

def test_fraude_exige_une_declaration(analyste, op_contestee, declaration_pour):
    r = emettre(analyste, op_contestee, motif="FRAUDE")
    assert r.status_code == 422 and r.json()["type"].endswith("/declaration-fraude-requise")
    assert emettre(analyste, op_contestee, motif="FRAUDE", declaration_fraude_id=declaration_pour(op_contestee)).status_code == 201

def test_contestation_obligatoire(analyste, op_non_contestee):
    assert emettre(analyste, op_non_contestee).status_code == 422

def test_un_seul_impaye_ouvert(analyste, op_contestee):
    assert emettre(analyste, op_contestee).status_code == 201
    assert emettre(analyste, op_contestee).status_code == 409
```
